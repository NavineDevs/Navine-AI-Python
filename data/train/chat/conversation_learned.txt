import re
from collections import Counter

def count_repeated_words(paragraph):
    # Remove punctuation marks at the end of words
    paragraph = re.sub(r'\b(\w+)[.,?!;:]+\B', r'\1', paragraph)
    
    # Split paragraph into words
    words = paragraph.lower().split()

    # Count repeated words
    word_count = Counter(words)

def calculate_modulo(x, y):
    """
    This function takes two positive integers, x and y, as input and calculates the modulo of x divided by y.
    The function returns the result of the modulo calculation.
    """

        raise ValueError("The input arguments must be positive integers.")

    if y == 0:
        raise ZeroDivisionError("Cannot divide by zero.")

        x -= y

import re
from collections import Counter

def count_repeated_words(paragraph):
    # Remove punctuation marks at the end of words
    paragraph = re.sub(r'\b(\w+)[.,?!;:]+\B', r'\1', paragraph)

    # Count repeated words

    if n <= 1:
        return n
    a, b = 0, 1
    for _ in range(2, n + 1):
        a, b = b, a + b
    return b

import math

def calculate_modulo(x, y):
    """
    This function takes two positive integers, x and y, as input and calculates the modulo of x divided by y.
    The function returns the result of the modulo calculation.
    """

    if not isinstance(x, int) or not isinstance(y, int) or x < 0 or y < 0:
        raise ValueError("The input arguments must be positive integers.")

    if y == 0:
        raise ZeroDivisionError("Cannot divide by zero.")

    while x >= y:
        x -= y

    return x

x = 5
y = 2

try:
    x_mod_y = calculate_modulo(x, y)
    print("The modulo of {} divided by {} is: {}".format(x, y, x_mod_y))

    if x_mod_y < 2:
        print("The modulo result is not a prime number.")
    else:
        is_prime = True
        for i in range(2, int(math.sqrt(x_mod_y)) + 1):
            if x_mod_y % i == 0:
                is_prime = False
                break
        
        if is_prime:
            print("The modulo result is a prime number.")
        else:
            print("The modulo result is not a prime number.")
except ValueError as e:
    print("Error:", e)
except ZeroDivisionError as e:
    print("Error:", e)

import re
from collections import Counter

def count_repeated_words(paragraph):
    # Remove punctuation marks at the end of words
    paragraph = re.sub(r'\b(\w+)[.,?!;:]+\B', r'\1', paragraph)
    
    # Split paragraph into words
    words = paragraph.lower().split()
    
    # Exclude words that contain 'e', are plural nouns, start with capital letter
    excluded_words = [word for word in words if 'e' in word or word.endswith('s') or word.istitle()]
    words = [word for word in words if word not in excluded_words]
    
    # Count repeated words
    word_count = Counter(words)
    
    return word_count

# Example paragraph

def fibonacci(n: int) -> int:
    if n <= 1:
        return n
    a, b = 0, 1
    for _ in range(2, n + 1):
        a, b = b, a + b
    return b

import math

def calculate_modulo(x, y):
    """
    This function takes two positive integers, x and y, as input and calculates the modulo of x divided by y.
    The function returns the result of the modulo calculation.
    """

    if not isinstance(x, int) or not isinstance(y, int) or x < 0 or y < 0:
        raise ValueError("The input arguments must be positive integers.")

    if y == 0:
        raise ZeroDivisionError("Cannot divide by zero.")

    while x >= y:
        x -= y

    return x

x = 5
y = 2

try:
    x_mod_y = calculate_modulo(x, y)
    print("The modulo of {} divided by {} is: {}".format(x, y, x_mod_y))

    if x_mod_y < 2:
        print("The modulo result is not a prime number.")
    else:
        is_prime = True
        for i in range(2, int(math.sqrt(x_mod_y)) + 1):
            if x_mod_y % i == 0:
                is_prime = False
                break
        
        if is_prime:
            print("The modulo result is a prime number.")
        else:
            print("The modulo result is not a prime number.")
except ValueError as e:
    print("Error:", e)
except ZeroDivisionError as e:
    print("Error:", e)

import re
from collections import Counter

def count_repeated_words(paragraph):
    # Remove punctuation marks at the end of words
    paragraph = re.sub(r'\b(\w+)[.,?!;:]+\B', r'\1', paragraph)
    
    # Split paragraph into words
    words = paragraph.lower().split()
    
    # Exclude words that contain 'e', are plural nouns, start with capital letter
    excluded_words = [word for word in words if 'e' in word or word.endswith('s') or word.istitle()]
    words = [word for word in words if word not in excluded_words]
    
    # Count repeated words
    word_count = Counter(words)
    
    return word_count

# Example paragraph

def fibonacci(n: int) -> int:
    if n <= 1:
        return n
    a, b = 0, 1
    for _ in range(2, n + 1):
        a, b = b, a + b
    return b

import math

def calculate_modulo(x, y):
    """
    This function takes two positive integers, x and y, as input and calculates the modulo of x divided by y.
    The function returns the result of the modulo calculation.
    """

    if not isinstance(x, int) or not isinstance(y, int) or x < 0 or y < 0:
        raise ValueError("The input arguments must be positive integers.")

    if y == 0:
        raise ZeroDivisionError("Cannot divide by zero.")

    while x >= y:
        x -= y

    return x

x = 5
y = 2

try:
    x_mod_y = calculate_modulo(x, y)
    print("The modulo of {} divided by {} is: {}".format(x, y, x_mod_y))

    if x_mod_y < 2:
        print("The modulo result is not a prime number.")
    else:
        is_prime = True
        for i in range(2, int(math.sqrt(x_mod_y)) + 1):
            if x_mod_y % i == 0:
                is_prime = False
                break
        
        if is_prime:
            print("The modulo result is a prime number.")
        else:
            print("The modulo result is not a prime number.")
except ValueError as e:
    print("Error:", e)
except ZeroDivisionError as e:
    print("Error:", e)

def fibonacci(n: int) -> int:
    if n <= 1:
        return n
    a, b = 0, 1
    for _ in range(2, n + 1):
        a, b = b, a + b
    return b

import math

def calculate_modulo(x, y):
    """
    This function takes two positive integers, x and y, as input and calculates the modulo of x divided by y.
    The function returns the result of the modulo calculation.
    """

    if not isinstance(x, int) or not isinstance(y, int) or x < 0 or y < 0:
        raise ValueError("The input arguments must be positive integers.")

    if y == 0:
        raise ZeroDivisionError("Cannot divide by zero.")

    while x >= y:
        x -= y

    return x

x = 5
y = 2

try:
    x_mod_y = calculate_modulo(x, y)
    print("The modulo of {} divided by {} is: {}".format(x, y, x_mod_y))

    if x_mod_y < 2:
        print("The modulo result is not a prime number.")
    else:
        is_prime = True
        for i in range(2, int(math.sqrt(x_mod_y)) + 1):
            if x_mod_y % i == 0:
                is_prime = False
                break
        
        if is_prime:
            print("The modulo result is a prime number.")
        else:
            print("The modulo result is not a prime number.")
except ValueError as e:
    print("Error:", e)
except ZeroDivisionError as e:
    print("Error:", e)

    root = Path(folder)

            continue

            continue
