import re
from collections import Counter

def count_repeated_words(paragraph):
    # Remove punctuation marks at the end of words
    paragraph = re.sub(r'\b(\w+)[.,?!;:]+\B', r'\1', paragraph)
    
    # Split paragraph into words
    words = paragraph.lower().split()
    
    # Exclude words that contain 'e', are plural nouns, start with capital letter
    excluded_words = [word for word in words if 'e' in word or word.endswith('s') or word.istitle()]
    words = [word for word in words if word not in excluded_words]
    
    # Count repeated words
    word_count = Counter(words)
    
    return word_count

# Example paragraph
paragraph = "The cats are sleeping. The Cat is awake. The cat, the dog, and the mouse are friends. The dog is playing."

# Count repeated words
word_count = count_repeated_words(paragraph)
print(word_count)

import math

def calculate_modulo(x, y):
    """
    This function takes two positive integers, x and y, as input and calculates the modulo of x divided by y.
    The function returns the result of the modulo calculation.
    """

    if not isinstance(x, int) or not isinstance(y, int) or x < 0 or y < 0:
        raise ValueError("The input arguments must be positive integers.")

    if y == 0:
        raise ZeroDivisionError("Cannot divide by zero.")

    while x >= y:
        x -= y

    return x

x = 5

    x_mod_y = calculate_modulo(x, y)
    print("The modulo of {} divided by {} is: {}".format(x, y, x_mod_y))

        print("The modulo result is not a prime number.")
    else:
        is_prime = True
        for i in range(2, int(math.sqrt(x_mod_y)) + 1):
            if x_mod_y % i == 0:
                is_prime = False
                break
        
        if is_prime:
            print("The modulo result is a prime number.")
        else:
            print("The modulo result is not a prime number.")
except ValueError as e:
    print("Error:", e)
except ZeroDivisionError as e:
    print("Error:", e)

def fibonacci(n: int) -> int:
    if n <= 1:
        return n
    a, b = 0, 1
    for _ in range(2, n + 1):

import re
from collections import Counter

def count_repeated_words(paragraph):
    # Remove punctuation marks at the end of words
    paragraph = re.sub(r'\b(\w+)[.,?!;:]+\B', r'\1', paragraph)
    
    # Split paragraph into words
    words = paragraph.lower().split()
    
    # Exclude words that contain 'e', are plural nouns, start with capital letter
    excluded_words = [word for word in words if 'e' in word or word.endswith('s') or word.istitle()]
    words = [word for word in words if word not in excluded_words]
    
    # Count repeated words
    word_count = Counter(words)
    
    return word_count

def fibonacci(n: int) -> int:
    if n <= 1:
        return n
    a, b = 0, 1
    for _ in range(2, n + 1):
        a, b = b, a + b
    return b

import math

def calculate_modulo(x, y):
    """
    This function takes two positive integers, x and y, as input and calculates the modulo of x divided by y.
    The function returns the result of the modulo calculation.
    """

    if not isinstance(x, int) or not isinstance(y, int) or x < 0 or y < 0:
        raise ValueError("The input arguments must be positive integers.")

    if y == 0:
        raise ZeroDivisionError("Cannot divide by zero.")

    while x >= y:
        x -= y

    return x

x = 5
y = 2

try:
    x_mod_y = calculate_modulo(x, y)
    print("The modulo of {} divided by {} is: {}".format(x, y, x_mod_y))

    if x_mod_y < 2:
        print("The modulo result is not a prime number.")
    else:
        is_prime = True
        for i in range(2, int(math.sqrt(x_mod_y)) + 1):
            if x_mod_y % i == 0:
                is_prime = False
                break
        
        if is_prime:
            print("The modulo result is a prime number.")
        else:
            print("The modulo result is not a prime number.")
except ValueError as e:
    print("Error:", e)
except ZeroDivisionError as e:
    print("Error:", e)

    allowed = set('0123456789+-*/().% ')
    if not expr or any(ch not in allowed for ch in expr):
        raise ValueError('Only basic math expressions are allowed')
    return str(eval(expr, {"__builtins__": {}}, {}))


def tool_case(text: str, mode: str) -> str:
    if mode == 'upper':
        return text.upper()
    if mode == 'lower':
        return text.lower()
    if mode == 'title':
        return text.title()
    return text


def tool_word_count(text: str) -> str:
    words = [w for w in text.split() if w.strip()]
    return f'words={len(words)} chars={len(text)} lines={len(text.splitlines())}'


def tool_batch_rename(folder: str, prefix: str) -> str:
    root = Path(folder)
    if not root.is_dir():
        raise ValueError('Folder not found')
    changed = []
    for idx, path in enumerate(sorted(root.iterdir()), start=1):
        if not path.is_file():
            continue
        target = root / f'{prefix}_{idx:03d}{path.suffix.lower()}'
        if target.exists():
            continue
        path.rename(target)
        changed.append(target.name)
    return f'renamed {len(changed)} files'

    '1': ('Calculator', lambda: tool_calculator(input('Expression: ').strip())),
    '2': ('Uppercase text', lambda: tool_case(input('Text: '), 'upper')),

}


def run_batch(jobs):
    results = []
    for job in jobs:
        tool_id = str(job.get('tool', '')).strip()
        payload = job.get('input', '')
        name, fn = TOOLS[tool_id]
        if tool_id == '1':
            out = tool_calculator(str(payload))
        elif tool_id in {'2', '3'}:
            out = tool_case(str(payload), 'upper' if tool_id == '2' else 'lower')

        elif tool_id == '5':

            folder = parts[0].strip()
